def identify_service(port):
    return known_services.get(port, "Unknown Service")

#port identify_service(port):
def parse_port_list(port_string):
    ports = set()
    for part in port_string.split(','):
        if '-' in part:
            try:
                a, b = part.split("-", 1)
                start = int(a.strip()); end = int(b.strip())
                if start < 1:
                    start = 1
                if end > 65535:
                    end = 65535
                if start <= end:
                    ports.update(range(start, end + 1))
            except Exception:
                continue
        else:
            try:
                port = int(part.strip())
                if MIN_PORT <= port <= MAX_PORT:
                    ports.add(port)
            except Exception:
                continue
    return list(ports)
def grab_banner(ip, port, timeout=2):
    try:
        sock = socket.socket()
        sock.settimeout(timeout)
        sock.connect((ip, port))
        banner = sock.recv(1024).decode().strip()
        sock.close()
        return banner
    except Exception:
        return "No Banner"

#OS guess from TTL / Window Size
def os_guess(ttl, window_size):
    if ttl >= 128:
        os_type = "Windows"
    elif ttl >= 64:
        os_type = "Linux/Unix"
    else:
        os_type = "Unknown"
    
    if window_size == 8192:
        os_detail = "Linux"
    elif window_size == 65535:
        os_detail = "Windows"
    else:
        os_detail = "Unknown"
    
    return f"{os_type} ({os_detail})"

def generate_decoys(num_decoys):
    decoys = set()
    while len(decoys) < num_decoys:
        decoy_ip = ".".join(str(random.randint(1, 254)) for _ in range(4))
        decoys.add(decoy_ip)
    return list(decoys)

# SYN Probe Worker Section (uses sr() for multiple packets)
def syn_probe(target_ip, target_port, timeout, use_fragmentation, decoys):

    tcp_layer = TCP(sport=RandShort(), dport=target_port, flags="S", seq=1000)
    packet = ip_layer / tcp_layer

    if use_fragmentation:
        packet = fragment(packet, fragsize=8)

    #If decoys are used, send packets from decoy IPs
    if decoys:
        for decoy in decoys:
            decoy_ip_layer = IP(src=decoy, dst=target_ip)
            decoy_packet = decoy_ip_layer / tcp_layer
            send(decoy_packet, verbose=0)

    response = sr1(packet, timeout=timeout, verbose=0)

    with results_lock:
        if response is None:
            filtered_results.append(target_port)

                                                     dport=target_port,
                                                     flags="R",
                                                     seq=response.getlayer(TCP).ack)
                send(rst_packet, verbose=0)

def output_detailed_results(results):
    # Group by target and service
    grouped = {}
    for ent in results:
        if isinstance(ent, dict):
            tgt = ent.get("target", "")
            svc = ent.get("service", "Unknown Service")
            grouped.setdefault(tgt, {})
            grouped[tgt].setdefault(svc, [])
            grouped[tgt][svc].append(ent)
        elif isinstance(ent, int):
            # If ent is just a port number, group under Unknown target/service
            grouped.setdefault("Unknown Target", {})
            grouped["Unknown Target"].setdefault("Unknown Service", [])
            grouped["Unknown Target"]["Unknown Service"].append({"port": ent})
    for tgt, svcs in grouped.items():
        print(colored(f"\n=== {tgt} ===", "cyan"))
        for svc, entries in svcs.items():
            header = f"{svc} ({len(entries)} port{'s' if len(entries)!=1 else ''})"
            print(colored(header, "magenta"))
            ports_sorted = sorted([ent["port"] for ent in entries if "port" in ent])
            ports_line = ", ".join(colored(str(p), "green") for p in ports_sorted)
            print("  " + ports_line)
            for ent in entries:
                b = ent.get("banner","") if isinstance(ent, dict) else ""
                if b:
                    short = (b[:180] + "...") if len(b) > 180 else b
                    print("    " + colored(f"banner[{ent['port']}]: ", "cyan") + short)

def main():

    parser.add_argument("-p", "--ports", required=True, help="Ports like '22,80,443' or ranges '1-1024' or mix")

    parser.add_argument("-f", "--format", choices=["table", "json"], default="table", help="Output format for terminal")
    parser.add_argument("-o", "--output", help="Write open-only results to JSON (and CSV alongside) if provided")
    parser.add_argument("-v", "--verbose", action="store_true", help="Verbose mode (prints extra info)")

    args = parser.parse_args()
    target_list = [t.strip() for t in args.targets.split(",") if t.strip()]
    # Enforce CIDR/range restriction for targets
    for t in target_list:
        if "/" in t or "-" in t:
            print(colored(f"[!] CIDR notation or IP ranges are not supported for targets: '{t}'", "red"))
            sys.exit(1)

            time.sleep(delay)

    # reduce aggressive concurrency on localhost to avoid overwhelming the stack
